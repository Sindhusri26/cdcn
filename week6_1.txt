%{
#include <stdio.h>

int yylex(void);
int yyerror(char *s);
%}

%token NUMBER

%%

E : E '+' T
  | E '-' T
  | T
  ;

T : T '*' F
  | T '/' F
  | F
  ;

F : '(' E ')'
  | NUMBER
  ;

%%

int yylex(void)
{
    int c;

    do {
        c = getchar();
    } while (c == ' ' || c == '\t' || c == '\n' || c == '\r');

    if (c == EOF)
        return 0;

    if (c >= '0' && c <= '9')
    {
        do {
            c = getchar();
        } while (c >= '0' && c <= '9');

        if (c != EOF)
            ungetc(c, stdin);

        return NUMBER;
    }

    return c;
}

int yyerror(char *s)
{
    printf("Parser error: %s\n", s);
    return 0;
}

int main(void)
{
    printf("Enter arithmetic expression: ");

    if (yyparse() == 0)
        printf("Valid expression\n");

    return 0;
}
