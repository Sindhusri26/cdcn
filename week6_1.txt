grammar:
E → E + T | T
T → T * F | F
F → ( E ) | id

Step 1 — YACC file
Create:ast.y




PROGRAM:

%{
#include <stdio.h>
#include <stdlib.h>

struct Node
{
    char value;
    struct Node *left;
    struct Node *right;
};

struct Node *root;

struct Node* node(char value, struct Node *left, struct Node *right)
{
    struct Node *n = malloc(sizeof(struct Node));
    n->value = value;
    n->left = left;
    n->right = right;
    return n;
}

void print(struct Node *r, int space)
{
    if (r == NULL)
        return;

    space += 5;

    print(r->right, space);

    printf("\n");
    for (int i = 5; i < space; i++)
        printf(" ");

    printf("%c\n", r->value);

    print(r->left, space);
}
%}

%union
{
    struct Node *ptr;
}

%token <ptr> ID
%type <ptr> E T F

%%

E : E '+' T   { $$ = node('+', $1, $3); root = $$; }
  | T         { $$ = $1; root = $$; }
  ;

T : T '*' F   { $$ = node('*', $1, $3); }
  | F         { $$ = $1; }
  ;

F : '(' E ')' { $$ = $2; }
  | ID        { $$ = $1; }
  ;

%%

int yylex()
{
    char c = getchar();

    if (c == 'i')
    {
        yylval.ptr = node('i', NULL, NULL);
        return ID;
    }

    return c;
}

int yyerror(char *s)
{
    printf("Invalid expression\n");
    return 0;
}

int main()
{
    printf("Enter expression using i for id: ");
    yyparse();

    printf("\nAbstract Syntax Tree:\n");
    print(root, 0);

    return 0;
}


1. Open PowerShell as Administrator
- Search PowerShell in Windows.
- Right-click → Run as administrator.
2. Install WSL + Ubuntu
wsl --install -d Ubuntu

3. Restart the laptop if Windows asks you to.
4. Open Ubuntu
- Search Ubuntu in the Start menu.
- Open it.
- Wait for installation to finish.
5. Create Ubuntu username and password
- Enter any username.
- Enter a password.
- ⚠️ While typing the password, nothing will appear. That's normal.
Install YACC and required tools
6. Update Ubuntu
sudo apt update

7. Install YACC/Bison, Flex and GCC
sudo apt install bison flex gcc

When it asks:
Do you want to continue? [Y/n]

Type:
Y

8. Check installation
yacc --version

If it shows a version, YACC is installed.
Run the YACC program
9. Create a folder
mkdir compiler
cd compiler

10. Create the YACC file
nano ast.y

11. Paste the YACC code into it.
12. Save:
- Ctrl + O
- Enter
- Ctrl + X
13. Generate the parser
yacc -d ast.y

14. Compile
gcc y.tab.c -o ast

15. Run
./ast

16. Enter input
i+i*i

17. You will get the Abstract Syntax Tree.
