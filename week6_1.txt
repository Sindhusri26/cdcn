%{
#include <stdio.h>
#include <stdlib.h>

int yylex(void);
int yyerror(const char *s);

struct Node
{
    char value;
    struct Node *left;
    struct Node *right;
};

struct Node *root = NULL;

struct Node* node(char value, struct Node *left, struct Node *right)
{
    struct Node *n = malloc(sizeof(struct Node));

    n->value = value;
    n->left = left;
    n->right = right;

    return n;
}

void print(struct Node *r, int space)
{
    if(r == NULL)
        return;

    space += 5;

    print(r->right, space);

    printf("\n");

    for(int i = 5; i < space; i++)
        printf(" ");

    printf("%c\n", r->value);

    print(r->left, space);
}
%}

%union
{
    struct Node *ptr;
}

%token <ptr> ID
%type <ptr> E T F

%%

E : E '+' T
    {
        $$ = node('+', $1, $3);
        root = $$;
    }
  | E '-' T
    {
        $$ = node('-', $1, $3);
        root = $$;
    }
  | T
    {
        $$ = $1;
        root = $$;
    }
  ;

T : T '*' F
    {
        $$ = node('*', $1, $3);
    }
  | T '/' F
    {
        $$ = node('/', $1, $3);
    }
  | F
    {
        $$ = $1;
    }
  ;

F : '(' E ')'
    {
        $$ = $2;
    }
  | ID
    {
        $$ = $1;
    }
  ;

%%

int yylex(void)
{
    int c;

    do
    {
        c = getchar();
    }
    while(c == ' ' || c == '\t' || c == '\n');

    if(c == EOF)
        return 0;

    if(c == 'i')
    {
        yylval.ptr = node('i', NULL, NULL);
        return ID;
    }

    return c;
}

int yyerror(const char *s)
{
    printf("Invalid expression\n");
    return 0;
}

    }

    return 0;
}

#include <iostream>
#include <string>
using namespace std;

struct Node
{
    char value;
    Node *left, *right;
};

Node* node(char value, Node *left = NULL, Node *right = NULL)
{
    Node *n = new Node;
    n->value = value;
    n->left = left;
    n->right = right;
    return n;
}

string s;
int pos = 0;

Node* E();
Node* T();
Node* F();

Node* E()
{
    Node *n = T();

    while(s[pos] == '+' || s[pos] == '-')
    {
        char op = s[pos++];
        n = node(op, n, T());
    }

    return n;
}

Node* T()
{
    Node *n = F();

    while(s[pos] == '*' || s[pos] == '/')
    {
        char op = s[pos++];
        n = node(op, n, F());
    }

    return n;
}

Node* F()
{
    if(s[pos] == '(')
    {
        pos++;
        Node *n = E();

        if(s[pos] == ')')
            pos++;
        else
            return NULL;

        return n;
    }

    if(s[pos] == 'i')
    {
        pos++;
        return node('i');
    }

    return NULL;
}

void print(Node *r, int space)
{
    if(r == NULL)
        return;

    space += 5;

    print(r->right, space);

    cout << endl;

    for(int i = 5; i < space; i++)
        cout << " ";

    cout << r->value << endl;

    print(r->left, space);
}

int main()
{
    cout << "Enter expression using i for identifier: ";
    cin >> s;

    Node *root = E();

    if(root != NULL && pos == s.length())
    {
        cout << "\nValid expression\n";
        cout << "\nAbstract Syntax Tree:\n";
        print(root, 0);
    }
    else
    {
        cout << "\nInvalid expression\n";
    }

    return 0;
}
