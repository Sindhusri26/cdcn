#include <iostream>
#include <cstdlib>
using namespace std;

int main()
{
    int ch;

    cout << "===== BASIC NETWORKING COMMANDS =====\n";

    cout << "\n1.  IP Configuration";
    cout << "\n2.  Detailed IP Configuration";
    cout << "\n3.  Ping Google";
    cout << "\n4.  Ping Localhost";
    cout << "\n5.  Trace Route";
    cout << "\n6.  Path Ping";
    cout << "\n7.  DNS Lookup";
    cout << "\n8.  Host Name";
    cout << "\n9.  MAC Address";
    cout << "\n10. ARP Table";
    cout << "\n11. Network Statistics";
    cout << "\n12. All Network Connections";
    cout << "\n13. Routing Table";
    cout << "\n14. Wi-Fi Interface";
    cout << "\n15. Wi-Fi Profiles";
    cout << "\n16. Exit";

    while (true)
    {
        cout << "\n\nEnter your choice: ";
        cin >> ch;

        if (cin.fail())
        {
            cin.clear();
            cin.ignore(1000, '\n');
            cout << "Enter a number from 1 to 16.\n";
            continue;
        }

        switch (ch)
        {
            case 1:
                cout << "\nIP Configuration\n";
                system("ipconfig");
                break;

            case 2:
                cout << "\nDetailed IP Configuration\n";
                system("ipconfig /all");
                break;

            case 3:
                cout << "\nPing Google\n";
                system("ping -n 4 google.com");
                break;

            case 4:
                cout << "\nPing Localhost\n";
                system("ping -n 4 127.0.0.1");
                break;

            case 5:
                cout << "\nTrace Route to Google\n";
                system("tracert -h 5 google.com");
                break;

            case 6:
                cout << "\nPath Ping to Google\n";
                system("pathping -q 2 -h 5 google.com");
                break;

            case 7:
                cout << "\nDNS Lookup\n";
                system("nslookup google.com");
                break;

            case 8:
                cout << "\nComputer Host Name\n";
                system("hostname");
                break;

            case 9:
                cout << "\nMAC Address\n";
                system("getmac");
                break;

            case 10:
                cout << "\nARP Table\n";
                system("arp -a");
                break;

            case 11:
                cout << "\nNetwork Statistics\n";
                system("netstat");
                break;

            case 12:
                cout << "\nAll Network Connections\n";
                system("netstat -a");
                break;

            case 13:
                cout << "\nRouting Table\n";
                system("route print");
                break;

            case 14:
                cout << "\nWi-Fi Interface Information\n";
                system("netsh wlan show interfaces");
                break;

            case 15:
                cout << "\nSaved Wi-Fi Profiles\n";
                system("netsh wlan show profiles");
                break;

            case 16:
                cout << "\nExiting program...\n";
                return 0;

            default:
                cout << "Invalid choice! Enter 1 to 16.";
        }

        cout << "\n----------------------------------------";
    }

    return 0;
}



INDIVIDUAL COMMANDS:
| No. | Command | Purpose |
|---:|---|---|
| 1 | `ipconfig` | Displays IP address and basic network configuration |
| 2 | `ipconfig /all` | Displays detailed IP and network adapter information |
| 3 | `ping google.com` | Checks connectivity with a remote host |
| 4 | `ping 127.0.0.1` | Tests the local computer's TCP/IP connection |
| 5 | `tracert google.com` | Displays the route taken to reach a destination |
| 6 | `pathping google.com` | Shows route information and packet loss |
| 7 | `nslookup google.com` | Displays DNS information for a domain |
| 8 | `hostname` | Displays the computer's host name |
| 9 | `getmac` | Displays the MAC address of network adapters |
| 10 | `arp -a` | Displays the ARP table |
| 11 | `netstat` | Displays active network connections |
| 12 | `netstat -a` | Displays all active connections and listening ports |
| 13 | `route print` | Displays the computer's routing table |
| 14 | `netsh wlan show interfaces` | Displays current Wi-Fi connection information |
| 15 | `netsh wlan show profiles` | Displays saved Wi-Fi network profiles |
