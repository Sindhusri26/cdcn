#include <bits/stdc++.h>
using namespace std;

map<char, set<char>> FIRST, FOLLOW;
map<char, vector<string>> prod;
set<char> nonTerminals;

set<char> first(string s)
{
    set<char> res;

    if(s.empty() || s == "#")
    {
        res.insert('#');
        return res;
    }

    bool eps = true;

    for(char c : s)
    {
        if(c == '#')
            continue;

        if(!nonTerminals.count(c))
        {
            res.insert(c);
            eps = false;
            break;
        }

        for(char x : FIRST[c])
            if(x != '#')
                res.insert(x);

        if(!FIRST[c].count('#'))
        {
            eps = false;
            break;
        }
    }

    if(eps)
        res.insert('#');

    return res;
}

int main()
{
    int n;
    string s;
    char start;

    cout << "Enter number of productions: ";
    cin >> n;

    cout << "Enter grammar rules (example: E->TX):\n";

    for(int i = 0; i < n; i++)
    {
        cin >> s;

        char lhs = s[0];

        if(i == 0)
            start = lhs;

        nonTerminals.insert(lhs);

        string rhs = s.substr(3);
        string cur;

        for(char c : rhs)
        {
            if(c == '|')
            {
                prod[lhs].push_back(cur);
                cur = "";
            }
            else
                cur += c;
        }

        prod[lhs].push_back(cur);
    }

    bool changed = true;

    while(changed)
    {
        changed = false;

        for(auto &[A, rules] : prod)
        {
            for(string p : rules)
            {
                set<char> f = first(p);

                for(char x : f)
                {
                    if(!FIRST[A].count(x))
                    {
                        FIRST[A].insert(x);
                        changed = true;
                    }
                }
            }
        }
    }

    FOLLOW[start].insert('$');

    changed = true;

    while(changed)
    {
        changed = false;

        for(auto &[A, rules] : prod)
        {
            for(string p : rules)
            {
                for(int i = 0; i < p.size(); i++)
                {
                    char B = p[i];

                    if(!nonTerminals.count(B))
                        continue;

                    string beta = p.substr(i + 1);
                    set<char> f = first(beta);

                    for(char x : f)
                    {
                        if(x != '#' && !FOLLOW[B].count(x))
                        {
                            FOLLOW[B].insert(x);
                            changed = true;
                        }
                    }

                    if(beta.empty() || f.count('#'))
                    {
                        for(char x : FOLLOW[A])
                        {
                            if(!FOLLOW[B].count(x))
                            {
                                FOLLOW[B].insert(x);
                                changed = true;
                            }
                        }
                    }
                }
            }
        }
    }

    cout << "\nFIRST SETS:\n";

    for(char c : nonTerminals)
    {
        cout << "FIRST(" << c << ") = { ";
        for(char x : FIRST[c])
            cout << x << " ";
        cout << "}\n";
    }

    cout << "\nFOLLOW SETS:\n";

    for(char c : nonTerminals)
    {
        cout << "FOLLOW(" << c << ") = { ";
        for(char x : FOLLOW[c])
            cout << x << " ";
        cout << "}\n";
    }

    return 0;
}
