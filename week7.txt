%{
#include <stdio.h>
#include <stdlib.h>

int yylex(void);
int yyerror(const char *s);

int result;
%}

%union
{
    int num;
}

%token <num> NUMBER
%type <num> E

%left '+' '-'
%left '*' '/'
%right UMINUS

%%

input : E
        {
            result = $1;
        }
      ;

E : E '+' E
    {
        $$ = $1 + $3;
    }
  | E '-' E
    {
        $$ = $1 - $3;
    }
  | E '*' E
    {
        $$ = $1 * $3;
    }
  | E '/' E
    {
        if($3 == 0)
        {
            printf("Division by zero\n");
            YYERROR;
        }

        $$ = $1 / $3;
    }
  | '-' E %prec UMINUS
    {
        $$ = -$2;
    }
  | '(' E ')'
    {
        $$ = $2;
    }
  | NUMBER
    {
        $$ = $1;
    }
  ;

%%

int yylex(void)
{
    int c;

    do
    {
        c = getchar();
    }
    while(c == ' ' || c == '\t' || c == '\n');

    if(c == EOF)
        return 0;

    if(c >= '0' && c <= '9')
    {
        yylval.num = 0;

        do
        {
            yylval.num = yylval.num * 10 + c - '0';
            c = getchar();
        }
        while(c >= '0' && c <= '9');

        if(c != EOF)
            ungetc(c, stdin);

        return NUMBER;
    }

    return c;
}

int yyerror(const char *s)
{
    printf("Invalid expression\n");
    return 0;
}

int main()
{
    printf("Enter arithmetic expression: ");

    if(yyparse() == 0)
    {
        printf("Valid expression\n");
        printf("Result = %d\n", result);
    }

    return 0;
}
