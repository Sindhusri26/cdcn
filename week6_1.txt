%{
#include <stdio.h>
#include <stdlib.h>

int yylex(void);
int yyerror(const char *s);

struct Node
{
    char value;
    struct Node *left;
    struct Node *right;
};

struct Node *root = NULL;

struct Node* node(char value, struct Node *left, struct Node *right)
{
    struct Node *n = malloc(sizeof(struct Node));

    n->value = value;
    n->left = left;
    n->right = right;

    return n;
}

void print(struct Node *r, int space)
{
    if(r == NULL)
        return;

    space += 5;

    print(r->right, space);

    printf("\n");

    for(int i = 5; i < space; i++)
        printf(" ");

    printf("%c\n", r->value);

    print(r->left, space);
}
%}

%union
{
    struct Node *ptr;
}

%token <ptr> ID
%type <ptr> E T F

%%

E : E '+' T
    {
        $$ = node('+', $1, $3);
        root = $$;
    }
  | E '-' T
    {
        $$ = node('-', $1, $3);
        root = $$;
    }
  | T
    {
        $$ = $1;
        root = $$;
    }
  ;

T : T '*' F
    {
        $$ = node('*', $1, $3);
    }
  | T '/' F
    {
        $$ = node('/', $1, $3);
    }
  | F
    {
        $$ = $1;
    }
  ;

F : '(' E ')'
    {
        $$ = $2;
    }
  | ID
    {
        $$ = $1;
    }
  ;

%%

int yylex(void)
{
    int c;

    do
    {
        c = getchar();
    }
    while(c == ' ' || c == '\t' || c == '\n');

    if(c == EOF)
        return 0;

    if(c == 'i')
    {
        yylval.ptr = node('i', NULL, NULL);
        return ID;
    }

    return c;
}

int yyerror(const char *s)
{
    printf("Invalid expression\n");
    return 0;
}

int main()
{
    printf("Enter expression using i for identifier: ");

    if(yyparse() == 0)
    {
        printf("\nValid expression\n");
        printf("\nAbstract Syntax Tree:\n");
        print(root, 0);
    }

    return 0;
}
